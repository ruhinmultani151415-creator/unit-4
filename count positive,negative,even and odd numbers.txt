//count positive,negative,even and odd numbers
#include <stdio.h>

int main()
{
    int a[100], n, i;
    int pos = 0, neg = 0, even = 0, odd = 0;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);

        if(a[i] > 0)
            pos++;
        else if(a[i] < 0)
            neg++;

        if(a[i] % 2 == 0)
            even++;
        else
            odd++;
    }

    printf("Positive = %d\n", pos);
    printf("Negative = %d\n", neg);
    printf("Even = %d\n", even);
    printf("Odd = %d", odd);

    return 0;
}
